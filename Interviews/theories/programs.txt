4. Find the Longest Substring Without Repeating Characters

Explanation: The answer is "abc", with the length of 3.

Input: s = "pwwkew"
Output: 3

var lengthOfLongestSubstring = function(s) {
    if(s.length === 0) {
        return 0;
    }
    if(s.length === 1) {
        return 1;
    }
    let set = new Set();
    var left = 0;
    var max = 0;
    for(let i=0;i<s.length;i++) {
        while(set.has(s.charAt(i))) {
            set.delete(s.charAt(left));
            left++;
        }
        set.add(s.charAt(i));
        max = Math.max(max, i-left+1);
    }
    return max;
};

var firstUniqChar = function(s) {
    if(s.length === 1) {
        return 0;
    }
    const map = new Map();
    for(let i=0;i<s.length;i++) {
        map.set(s[i], (map.get(s[i]) || 0) + 1);
    }
    
    for(let i=0;i<s.length;i++) {
        if(map.get(s[i]) === 1) {
            return i;
        }
    }
    return -1;
};

function permute(str, l = 0, r = str.length - 1) {
    if (l === r) {
        console.log(str); // Print or store the permutation
        return;
    }
    for (let i = l; i <= r; i++) {
        str = swap(str, l, i); 
        permute(str, l + 1, r);
        str = swap(str, l, i); // Backtrack
    }
}

function swap(s, i, j) {
    let arr = s.split('');
    [arr[i], arr[j]] = [arr[j], arr[i]];
    return arr.join('');
}

20. Implement a Method to Compare Two Strings for Equality (Without Using .equals())
function areStringsEqual(str1, str2) {
    // First, check if the lengths are the same
    if (str1.length !== str2.length) {
        return false;  // Strings of different lengths are not equal
    }

    // Compare each character of both strings
    for (let i = 0; i < str1.length; i++) {
        if (str1[i] !== str2[i]) {
            return false;  // If any character is different, return false
        }
    }

    // If no differences were found, the strings are equal
    return true;
}

23. Check if a String is a Subsequence of Another String
Given two strings s and t, return true if s is a subsequence of t, or false otherwise.

A subsequence of a string is a new string that is formed from the original string by deleting some 
(can be none) of the characters without disturbing the relative positions of the remaining characters. 
(i.e., "ace" is a subsequence of "abcde" while "aec" is not).

Example 1:

Input: s = "abc", t = "ahbgdc"
Output: true
Example 2:
Input: s = "axc", t = "ahbgdc"
Output: false


var isSubsequence = function(s, t) {
    let i=0, j=0;
    while(i<s.length && j<t.length) {
        if(s[i] === t[j]) {
            i++;
        }
        j++
    }
    return i === s.length;
};

function uniqueCharacters(str) {
    let set = new Set();
    let duplicates = new Set();

    for (let char of str) {
        if (set.has(char)) {
            duplicates.add(char);
        } else {
            set.add(char);
        }
    }

    // Remove duplicate characters from set
    for (let char of duplicates) {
        set.delete(char);
    }

    return [...set]; // Convert Set to Array
}


25. Convert a Given String to Camel Case
function toCamelCase(str) {
    return str
        .toLowerCase()
        .split(/[^a-zA-Z0-9]+/) // Split by spaces, dashes, underscores, etc.
        .map((word, index) =>
            index === 0
                ? word // Keep the first word as lowercase
                : word.charAt(0).toUpperCase() + word.slice(1).toLowerCase()
        )
        .join('');
}


26. Count the Number of Times One String Occurs in Another
function countOccurrences(str, subStr) {
    let count = 0;
    let index = str.indexOf(subStr); // Find first occurrence

    while (index !== -1) { // Keep searching while found
        count++;
        index = str.indexOf(subStr, index + subStr.length); // Move past last found occurrence
    }

    return count;
}
console.log(countOccurrences("hello hello world hello", "hello")); // 3
console.log(countOccurrences("abababab", "aba")); // 2 (non-overlapping)
console.log(countOccurrences("abcabcabc", "abc")); // 3

29. Merge Two Strings Alternately
function mergeAlternately(str1, str2) {
    let result = '';
    let len1 = str1.length;
    let len2 = str2.length;
    let i = 0;

    // Loop until one of the strings is exhausted
    while (i < len1 || i < len2) {
        if (i < len1) result += str1[i];  // Add character from str1 if available
        if (i < len2) result += str2[i];  // Add character from str2 if available
        i++;
    }

    return result;
}

// Example usage:
console.log(mergeAlternately("abc", "123")); // "a1b2c3"
console.log(mergeAlternately("abcd", "12")); // "a1b2cd"
console.log(mergeAlternately("hello", "world")); // "hweolrllod"

function getSecondLargest(arr) {
    const n = arr.length;

    let largest = -1, secondLargest = -1;

    for (let i = 0; i < n; i++) {
        if (arr[i] > largest) {
            secondLargest = largest;
            largest = arr[i];
        }
        else if (arr[i] < largest && arr[i] > secondLargest) {
            secondLargest = arr[i];
        }
    }
    return secondLargest;
}

const flattenObject = (obj, parent) => {
    const resObj = {};
    const generateFlatObject = (obj, parent) => {
        for(let key in obj) {
            const newParent = parent+key;
            const value = obj[key];
            if(typeof value === 'object'){
                generateFlatObject(value, newParent+".")
            } else {
                resObj[newParent] = value
            }
        }
    }
    generateFlatObject(obj, parent);
    return resObj;
}
